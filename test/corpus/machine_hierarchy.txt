machine Workflow {
    events { Start; Pause; Resume; }
    emits { Trace { label: string; } }
    state Idle;
    parallel state Working {
        generation: i64;
        entry { emit Trace { label: "working" }; }
        region left {
            state Editing {
                draft: string;
                initial state Typing;
                state Saved;
            }
            initial Editing { draft: "new".to_upper() };
        }
        region right {
            initial state Waiting;
            state Sent;
        }
    }
    on Start: Idle => Working { generation: 1 }
    on Pause: Working => Idle;
    on Resume: Idle => Working history reenter;
    default { state }
}

================================================================================
Machine completion and eventless triggers
================================================================================

machine Delivery {
    events { Send; }
    state Ready;
    state Running {
        initial state Waiting;
        final state Sent;
        on Send: Waiting => Sent;
    }
    final state Finished;
    on Send: Ready => Running;
    on completion: Running => Finished;
    always: Ready => Finished when false;
    default { state }
}

           completion: i64, always: i64, reenter: i64) -> i64 {
    parallel + region + initial + final + completion + always + reenter
