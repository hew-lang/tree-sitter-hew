================================================================================
Race composes with a scope deadline and fallible return
================================================================================

fn work(value: i64) -> i64 fails Error {
    value
}

fn main() {
    scope within 2s {
        let winner = race { work(1), work(2), };
        let task = fork work(winner);
        await task
    } handle failure {
        println(failure);
    }
}

Prepared task selection and timer arms

fn main() {
    let first = fork work(1);
    let second = fork work(2);
    let selected = select {
        value from await first => value + await second,
        value from await second => value + await first,
        after 1s => 0,
    };
    println(selected);
}
