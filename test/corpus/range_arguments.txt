fn ranges(text: string, end: i64) {
    text.get(..);
    text.get(..end);
    text.get(..=end);
    text.get(end..);
    text.get(1..end);
    text.get(1..=end);
    text.get((end + 1)..);
    accept(range: ..end, other: 2);
    accept(.., 2);
    accept(if true { 1 } else { 2 }, end + 1);
}
