macro wrap {
    #[($label:literal)] $m:method fails => items {
        $m.vis fn $m.name $m.generics ($m.params) $m.ret $m.where {
            helper($m.args);
            $m.body
        }
    }
    #[()] $f:fn => items { #[tests.observed("wrapped")] $f }
    ($value:expr) => expr { $value + 1 }
}
#[wrappers.wrap("outer")]
#[wrap]
fn answer() -> i64 { wrap!(41) }
trait Answers {
    #[wrap]
    fn[suspends] answer(self) -> i64 { 42 }
}
